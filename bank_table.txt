CREATE TABLE EMPLOYEE(  
   employee_id   INTEGER,  
   name          VARCHAR(50),  
   address       VARCHAR(255),  
   phone_number  VARCHAR(15),  
   branch_id     INTEGER  
);  

CREATE TABLE BRANCH(  
   branch_id      INTEGER,  
   name           VARCHAR(100),  
   address        VARCHAR(255),  
   contact_number VARCHAR(15)  
);  

CREATE TABLE ACCOUNT(  
   account_number  INTEGER,  
   balance         DECIMAL(12, 2),  
   status          VARCHAR(20),  
   account_type    VARCHAR(50),  
   open_date       DATE,  
   branch_id       INTEGER  
);  

CREATE TABLE CHEQUE(  
   cheque_id      INTEGER,  
   cheque_number  VARCHAR(50),  
   amount         DECIMAL(12, 2),  
   status         VARCHAR(20),  
   issue_date     DATE,  
   account_number INTEGER  
);  

CREATE TABLE TRANSACTION(  
   transaction_id   INTEGER,  
   transaction_type VARCHAR(50),  
   amount          DECIMAL(12, 2),  
   date            DATE,  
   time            TIME,  
   status          VARCHAR(20),  
   account_number  INTEGER  
);  

CREATE TABLE CARD(  
   card_number  VARCHAR(20),  
   card_type    VARCHAR(20),  
   expiry_date  DATE,  
   cvv         INTEGER,  
   status       VARCHAR(20),  
   account_number INTEGER  
);  

CREATE TABLE ATM_TRANSACTION(  
   atmtransaction_id INTEGER,  
   amount           DECIMAL(12, 2),  
   location         VARCHAR(255),  
   date            DATE,  
   time            TIME,  
   card_number     VARCHAR(20)  
);  

CREATE TABLE CUSTOMER(  
   customer_id      INTEGER,  
   name            VARCHAR(100),  
   email           VARCHAR(100),  
   phone_number    VARCHAR(15),  
   registration_date DATE,  
   date_of_birth   DATE  
);  

CREATE TABLE LOAN(  
   loan_id       INTEGER,  
   amount        DECIMAL(12, 2),  
   loan_type     VARCHAR(50),  
   approval_date DATE,  
   interest_rate DECIMAL(5, 2),  
   status        VARCHAR(20),  
   customer_id   INTEGER  
);  

CREATE TABLE LOAN_PAYMENT(  
   payment_id    INTEGER,  
   amount        DECIMAL(12, 2),  
   status        VARCHAR(20),  
   payment_date  DATE,  
   loan_id       INTEGER  
);  

CREATE TABLE FIXED_DEPOSIT(  
   fd_id          INTEGER,  
   amount        DECIMAL(12, 2),  
   maturity_date DATE,  
   interest_rate DECIMAL(5, 2),  
   status        VARCHAR(20),  
   customer_id   INTEGER  
);  

CREATE TABLE CUSTOMER_SERVICE(  
   service_id       INTEGER,  
   issue           VARCHAR(255),  
   resolution_date DATE,  
   status          VARCHAR(20),  
   customer_id     INTEGER  
);

   PRIMARY KEY (account_number);

   ADD CONSTRAINT cheque_pk  
   PRIMARY KEY (cheque_id);

   ADD CONSTRAINT transaction_pk  
   PRIMARY KEY (transaction_id);  

ALTER TABLE CARD  
   ADD CONSTRAINT card_pk  
   PRIMARY KEY (card_number);

   ADD CONSTRAINT atm_transaction_pk  
   PRIMARY KEY (atmtransaction_id);

   ADD CONSTRAINT customer_pk  
   PRIMARY KEY (customer_id);

   ADD CONSTRAINT loan_pk  
   PRIMARY KEY (loan_id);

   ADD CONSTRAINT loan_payment_pk  
   PRIMARY KEY (payment_id);

   PRIMARY KEY (fd_id);

   ADD CONSTRAINT service_pk  
   PRIMARY KEY (service_id);

   ADD CONSTRAINT emp_branch_fk  
   FOREIGN KEY (branch_id)  
   REFERENCES BRANCH (branch_id);

   ADD CONSTRAINT account_branch_fk  
   FOREIGN KEY (branch_id)  
   REFERENCES BRANCH (branch_id);

   ADD CONSTRAINT cheque_account_fk  
   FOREIGN KEY (account_number)  
   REFERENCES ACCOUNT (account_number);

   ADD CONSTRAINT transaction_account_fk  
   FOREIGN KEY (account_number)  
   REFERENCES ACCOUNT (account_number);  

-- CARD to ACCOUNT  
ALTER TABLE CARD  
   ADD CONSTRAINT card_account_fk  
   FOREIGN KEY (account_number)  
   REFERENCES ACCOUNT (account_number);  

-- ATM_TRANSACTION to CARD  
ALTER TABLE ATM_TRANSACTION

   FOREIGN KEY (card_number)  
   REFERENCES CARD (card_number);  

-- LOAN to CUSTOMER  
ALTER TABLE LOAN  
   ADD CONSTRAINT loan_customer_fk  
   FOREIGN KEY (customer_id)  
   REFERENCES CUSTOMER (customer_id);

ALTER TABLE LOAN_PAYMENT  
   ADD CONSTRAINT payment_loan_fk  
   FOREIGN KEY (loan_id)  
   REFERENCES LOAN (loan_id);

   ADD CONSTRAINT fd_customer_fk
